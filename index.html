window.deleteItem = function (category, id) { if (!confirm("Voulez-vous vraiment supprimer cet élément ?")) return; if (!state[category]) return; var item = state[category].find(function (entry) { return entry.id === id; }); if (!item) return; // Si une vente est supprimée, restituer les articles au stock. if (category === "sales" && item.productId) { var product = state.products.find(function (p) { return p.id === item.productId; }); if (product && product.type !== "Service") { product.stock = number(product.stock) + number(item.qty); } } state[category] = state[category].filter(function (entry) { return entry.id !== id; }); saveState(); render(); notify( category === "sales" ? "Vente supprimée et stock rétabli." : "Élément supprimé." ); };
